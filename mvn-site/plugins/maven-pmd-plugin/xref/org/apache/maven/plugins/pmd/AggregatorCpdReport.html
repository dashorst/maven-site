19  package org.apache.maven.plugins.pmd;
20  
21  import javax.inject.Inject;
22  
23  import org.apache.maven.plugins.annotations.Mojo;
24  import org.apache.maven.plugins.pmd.exec.CpdServiceExecutor;
25  import org.codehaus.plexus.i18n.I18N;
26  
27  /**
28   * Creates a report for PMD's Copy/Paste Detector (CPD) tool in an <b>aggregator</b> project.
29   * It can also generate a cpd results file in any of these formats: xml, csv or txt.
30   *
31   * <p>See <a href="https://pmd.github.io/latest/pmd_userdocs_cpd.html">Finding duplicated code</a>
32   * for more details.
33   *
34   * @since 3.15.0
35   */
36  @Mojo(name = "aggregate-cpd", aggregator = true, threadSafe = true)
37  public class AggregatorCpdReport extends CpdReport {
38  
39      @Inject
40      public AggregatorCpdReport(I18N i18n, CpdServiceExecutor serviceExecutor) {
41          super(i18n, serviceExecutor);
42      }
43  
44      @Override
45      protected boolean isAggregator() {
46          return true;
47      }
48  }
