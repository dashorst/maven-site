256     public void testExecuteShouldAddOneModuleSetWithOneModuleInIt() throws Exception {
257         final MavenProject project = createProject("group", "artifact", "version", null);
258 
259         final MavenProject module = createProject("group", "module", "version", project);
260 
261         Artifact artifact = mock(Artifact.class);
262         final File moduleArtifactFile = File.createTempFile("junit", null, temporaryFolder);
263         when(artifact.getGroupId()).thenReturn("GROUPID");
264         when(artifact.getFile()).thenReturn(moduleArtifactFile);
265         module.setArtifact(artifact);
266 
267         final List<MavenProject> projects = new ArrayList<>();
268         projects.add(module);
269 
270         final AssemblerConfigurationSource configSource = mock(AssemblerConfigurationSource.class);
271         when(configSource.getReactorProjects()).thenReturn(projects);
272         when(configSource.getFinalName()).thenReturn("final-name");
273         when(configSource.getProject()).thenReturn(project);
274 
275         final Archiver archiver = mock(Archiver.class);
276         when(archiver.getDestFile()).thenReturn(new File("junk"));
277         when(archiver.getOverrideDirectoryMode()).thenReturn(0777);
278         when(archiver.getOverrideFileMode()).thenReturn(0777);
279 
280         final ModuleBinaries bin = new ModuleBinaries();
281         bin.setOutputFileNameMapping("artifact");
282         bin.setOutputDirectory("out");
283         bin.setFileMode("777");
284         bin.setUnpack(false);
285         bin.setIncludeDependencies(false);
286 
287         final ModuleSet ms = new ModuleSet();
288         ms.setBinaries(bin);
289 
290         final Assembly assembly = new Assembly();
291         assembly.setIncludeBaseDirectory(false);
292         assembly.addModuleSet(ms);
293 
294         when(dependencyResolver.resolveDependencySets(eq(assembly), eq(ms), eq(configSource), anyList()))
295                 .thenReturn(new LinkedHashMap<>());
296         DefaultAssemblyArchiverTest.setupInterpolators(configSource, module);
297 
298         this.phase.execute(assembly, archiver, configSource);
299 
300         // result of easymock migration, should be assert of expected result instead of verifying methodcalls
301         verify(configSource, atLeastOnce()).getFinalName();
302         verify(configSource, atLeastOnce()).getMavenSession();
303         verify(configSource, atLeastOnce()).getProject();
304         verify(configSource, atLeastOnce()).getReactorProjects();
305 
306         verify(archiver).addFile(moduleArtifactFile, "out/artifact", 511);
307         verify(archiver, atLeastOnce()).getDestFile();
308         verify(archiver).getOverrideDirectoryMode();
309         verify(archiver).getOverrideFileMode();
310         verify(archiver, times(2)).setFileMode(511);
311 
312         verify(dependencyResolver).resolveDependencySets(eq(assembly), eq(ms), eq(configSource), anyList());
313     }

732     private void verifyResultIs(final List<MavenProject> check, final Set<MavenProject> moduleProjects) {
733         boolean failed = false;
734 
735         final Set<MavenProject> checkTooMany = new HashSet<>(moduleProjects);
736         check.forEach(checkTooMany::remove);
737 
738         if (!checkTooMany.isEmpty()) {
739             failed = true;
740 
741             System.out.println("Unexpected projects in output: ");
742 
743             for (final MavenProject project : checkTooMany) {
744                 System.out.println(project.getId());
745             }
746         }
747 
748         final Set<MavenProject> checkTooFew = new HashSet<>(check);
749         checkTooFew.removeAll(moduleProjects);
750 
751         if (!checkTooFew.isEmpty()) {
752             failed = true;
753 
754             System.out.println("Expected projects missing from output: ");
755 
756             for (final MavenProject project : checkTooMany) {
757                 System.out.println(project.getId());
758             }
759         }
760 
761         if (failed) {
762             fail("See system output for more information.");
763         }
764     }

766     private MavenProject createProject(
767             final String groupId, final String artifactId, final String version, final MavenProject parentProject) {
768         final Model model = new Model();
769         model.setArtifactId(artifactId);
770         model.setGroupId(groupId);
771         model.setVersion(version);
772 
773         final MavenProject project = new MavenProject(model);
774 
775         File pomFile;
776         if (parentProject == null) {
777             final File basedir = temporaryFolder;
778             pomFile = new File(basedir, "pom.xml");
779         } else {
780             final File parentBase = parentProject.getBasedir();
781             pomFile = new File(parentBase, artifactId + "/pom.xml");
782 
783             parentProject.getModel().addModule(artifactId);
784             project.setParent(parentProject);
785         }
786 
787         project.setFile(pomFile);
788 
789         return project;
790     }
