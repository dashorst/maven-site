19  package org.apache.maven.plugins.assembly.io;
20  
21  import java.net.URL;
22  
23  /**
24   * classpath resource locator strategy.
25   */
26  class ClasspathResourceLocatorStrategy implements LocatorStrategy {
27  
28      /**
29       * Create instance.
30       */
31      ClasspathResourceLocatorStrategy() {}
32  
33      /** {@inheritDoc} */
34      public Location resolve(String locationSpecification, MessageHolder messageHolder) {
35          ClassLoader cloader = Thread.currentThread().getContextClassLoader();
36  
37          URL resource = cloader.getResource(locationSpecification);
38  
39          Location location = null;
40  
41          if (resource != null) {
42              location = new URLLocation(resource, locationSpecification, "location.", ".cpurl", true);
43          } else {
44              messageHolder.addMessage(
45                      "Failed to resolve classpath resource: " + locationSpecification + " from classloader: " + cloader);
46          }
47  
48          return location;
49      }
50  }
