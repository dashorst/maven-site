37  /**
38   * Tests {@link SelectorUtils}.
39   *
40   * @author Benjamin Bentmann

42  class SelectorUtilsTest {
43  
44      @Test
45      void parseList() {
46          List<String> includes = new ArrayList<>();
47          List<String> excludes = new ArrayList<>();
48  
49          SelectorUtils.parseList(null, includes, excludes);
50  
51          SelectorUtils.parseList(" 1.5, !1.4, 1.6+ ", includes, excludes);
52          assertThat(includes).containsExactly("1.5", "1.6+");
53          assertThat(excludes).containsExactly("1.4");
54      }
55  
56      @Test
57      void parseVersion() {
58          assertThat(SelectorUtils.parseVersion("1.6.0_12")).containsExactly(1, 6, 0, 12);
59          assertThat(SelectorUtils.parseVersion("1.6.0_12+")).containsExactly(1, 6, 0, 12);
60          assertThat(SelectorUtils.parseVersion("1.6.0_12-")).containsExactly(1, 6, 0, 12);
61      }
62  
63      @Test
64      void compareVersions() {
65          assertThat(SelectorUtils.compareVersions(Arrays.asList(1, 6), Arrays.asList(1, 6)))
66                  .isZero();
67  
68          assertThat(SelectorUtils.compareVersions(Arrays.asList(1, 5), Arrays.asList(1, 6)))
69                  .isNegative();
70          assertThat(SelectorUtils.compareVersions(Arrays.asList(1, 6), Arrays.asList(1, 5)))
71                  .isPositive();
72  
73          assertThat(SelectorUtils.compareVersions(Collections.singletonList(1), Arrays.asList(1, 6)))
74                  .isNegative();
75          assertThat(SelectorUtils.compareVersions(Arrays.asList(1, 6), Collections.singletonList(1)))
76                  .isPositive();
77      }
78  
79      @Test
80      void isMatchingJre() {
81  
82          assertThat(SelectorUtils.isJreVersion(Arrays.asList(1, 4, 2, 8), "1.5")).isFalse();
83          assertThat(SelectorUtils.isJreVersion(Arrays.asList(1, 5), "1.5")).isTrue();
84          assertThat(SelectorUtils.isJreVersion(Arrays.asList(1, 5, 9), "1.5")).isTrue();
85          assertThat(SelectorUtils.isJreVersion(Arrays.asList(1, 6), "1.5")).isFalse();
86  
87          assertThat(SelectorUtils.isJreVersion(Arrays.asList(1, 4, 2, 8), "1.5+"))
88                  .isFalse();
89          assertThat(SelectorUtils.isJreVersion(Arrays.asList(1, 5), "1.5+")).isTrue();
90          assertThat(SelectorUtils.isJreVersion(Arrays.asList(1, 5, 9), "1.5+")).isTrue();
91          assertThat(SelectorUtils.isJreVersion(Arrays.asList(1, 6), "1.5+")).isTrue();
92  
93          assertThat(SelectorUtils.isJreVersion(Arrays.asList(1, 4, 2, 8), "1.5-"))
94                  .isTrue();
95          assertThat(SelectorUtils.isJreVersion(Arrays.asList(1, 5), "1.5-")).isFalse();
96          assertThat(SelectorUtils.isJreVersion(Arrays.asList(1, 5, 9), "1.5-")).isFalse();
97          assertThat(SelectorUtils.isJreVersion(Arrays.asList(1, 6), "1.5-")).isFalse();
98  
99          assertThat(SelectorUtils.isJreVersion((String) null, "1.5")).isTrue();
100         assertThat(SelectorUtils.isJreVersion("", "1.5")).isTrue();
101     }
102 
103     @Test
104     void isMatchingToolchain() {
105         InvokerToolchain openJdk9 = new InvokerToolchain("jdk");
106 
107         ToolchainPrivateManager toolchainPrivateManager = mock(ToolchainPrivateManager.class);
108 
109         when(toolchainPrivateManager.isToolchains(eq("jdk"), any())).thenReturn(true);
110         assertThat(SelectorUtils.isToolchain(toolchainPrivateManager, Collections.singleton(openJdk9)))
111                 .isTrue();
112 
113         when(toolchainPrivateManager.isToolchains(eq("jdk"), any())).thenReturn(false);
114         assertThat(SelectorUtils.isToolchain(toolchainPrivateManager, Collections.singleton(openJdk9)))
115                 .isFalse();
116     }
117 
118     @Test
119     void mavenVersionForNotExistingMavenHomeThrowException() {
120         File mavenHome = new File("not-existing-path");
121 
122         assertThatCode(() -> SelectorUtils.getMavenVersion(mavenHome))
123                 .isExactlyInstanceOf(IllegalArgumentException.class)
124                 .hasMessage("Invalid Maven home installation directory: not-existing-path");
125     }
126 
127     @Test
128     void mavenVersionFromMavenHome() throws Exception {
129         File mavenHome = new File(System.getProperty("maven.home"));
130 
131         String mavenVersion = SelectorUtils.getMavenVersion(mavenHome);
132 
133         assertThat(mavenVersion).isNotBlank();
134     }
