221     public void execute() throws MojoExecutionException {
222 
223         // Run only at the execution root
224         if (runOnlyAtExecutionRoot && !isThisTheExecutionRoot()) {
225             getLog().info("Skipping the announcement mail in this project because it's not the Execution Root");
226         } else {
227             File file = new File(announcementDirectory, announcementFile);
228 
229             ConsoleLogger logger = new ConsoleLogger(Logger.LEVEL_INFO, "base");
230 
231             if (getLog().isDebugEnabled()) {
232                 logger.setThreshold(Logger.LEVEL_DEBUG);
233             }
234 
235             mailer.enableLogging(logger);
236 
237             mailer.setSmtpHost(getSmtpHost());
238 
239             mailer.setSmtpPort(getSmtpPort());
240 
241             mailer.setSslMode(sslMode, startTls);
242 
243             if (username != null) {
244                 mailer.setUsername(username);
245             }
246 
247             if (password != null) {
248                 mailer.setPassword(password);
249             }
250 
251             mailer.initialize();
252 
253             if (getLog().isDebugEnabled()) {
254                 getLog().debug("fromDeveloperId: " + getFromDeveloperId());
255             }
256 
257             if (file.isFile()) {
258                 getLog().info("Connecting to Host: " + getSmtpHost() + ":" + getSmtpPort());
259 
260                 sendMessage();
261             } else {
262                 throw new MojoExecutionException("Announcement file " + file + " not found...");
263             }
264         }
265     }

267     /**
268      * Send the email.
269      *
270      * @throws MojoExecutionException if the mail could not be sent
271      */
272     protected void sendMessage() throws MojoExecutionException {
273         File file = new File(announcementDirectory, announcementFile);
274         String email = "";
275         final MailSender ms = getActualMailSender();
276         final String fromName = ms.getName();
277         final String fromAddress = ms.getEmail();
278         if (fromAddress == null || fromAddress.isEmpty()) {
279             throw new MojoExecutionException("Invalid mail sender: name and email is mandatory (" + ms + ").");
280         }
281         getLog().info("Using this sender for email announcement: " + fromAddress + " < " + fromName + " > ");
282         try {
283             MailMessage mailMsg = new MailMessage();
284             mailMsg.setSubject(getSubject());
285             mailMsg.setContent(readAnnouncement(file));
286             mailMsg.setContentType(this.mailContentType);
287             mailMsg.setFrom(fromAddress, fromName);
288 
289             for (Object o1 : getToAddresses()) {
290                 email = o1.toString();
291                 getLog().info("Sending mail to " + email + "...");
292                 mailMsg.addTo(email, "");
293             }
294 
295             if (getCcAddresses() != null) {
296                 for (Object o : getCcAddresses()) {
297                     email = o.toString();
298                     getLog().info("Sending cc mail to " + email + "...");
299                     mailMsg.addCc(email, "");
300                 }
301             }
302 
303             if (getBccAddresses() != null) {
304                 for (Object o : getBccAddresses()) {
305                     email = o.toString();
306                     getLog().info("Sending bcc mail to " + email + "...");
307                     mailMsg.addBcc(email, "");
308                 }
309             }
310 
311             mailer.send(mailMsg);
312             getLog().info("Sent...");
313         } catch (MailSenderException e) {
314             throw new MojoExecutionException("Failed to send email < " + email + " >", e);
315         }
316     }

318     /**
319      * Read the content of the generated announcement file.
320      *
321      * @param file the file to be read
322      * @return Return the announcement text
323      * @throws MojoExecutionException if the file could not be found, or if the encoding is unsupported
324      */
325     protected String readAnnouncement(File file) throws MojoExecutionException {
326         try {
327             if (templateEncoding == null || templateEncoding.isEmpty()) {
328                 templateEncoding = Charset.defaultCharset().displayName();
329                 getLog().warn("File encoding has not been set, using platform encoding '" + templateEncoding
330                         + "', i.e. build is platform dependent!");
331             }
332 
333             try (InputStreamReader reader = new InputStreamReader(new FileInputStream(file), templateEncoding)) {
334                 return IOUtil.toString(reader);
335             }
336         } catch (FileNotFoundException fnfe) {
337             throw new MojoExecutionException("File not found. " + file);
338         } catch (UnsupportedEncodingException uee) {
339             throw new MojoExecutionException("Unsupported encoding: '" + templateEncoding + "'");
340         } catch (IOException ioe) {
341             throw new MojoExecutionException("Failed to read the announcement file.", ioe);
342         }
343     }

345     /**
346      * Returns the identify of the mail sender according to the plugin's configuration:
347      * <ul>
348      * <li>if the {@code mailSender} parameter is set, it is returned</li>
349      * <li>if no {@code fromDeveloperId} is set, the first developer in the list is returned</li>
350      * <li>if a {@code fromDeveloperId} is set, the developer with that id is returned</li>
351      * <li>if the developers list is empty or if the specified id does not exist, an exception is thrown</li>
352      * </ul>
353      *
354      * @return the mail sender to use
355      * @throws MojoExecutionException if the mail sender could not be retrieved
356      */
357     protected MailSender getActualMailSender() throws MojoExecutionException {
358         if (senderString != null) {
359             try {
360                 InternetAddress ia = new InternetAddress(senderString, true);
361                 return new MailSender(ia.getPersonal(), ia.getAddress());
362             } catch (AddressException e) {
363                 throw new MojoExecutionException("Invalid value for change.sender: ", e);
364             }
365         }
366         if (mailSender != null && mailSender.getEmail() != null) {
367             return mailSender;
368         } else if (from == null || from.isEmpty()) {
369             throw new MojoExecutionException("The <developers> section in your pom should not be empty. "
370                     + "Add a <developer> entry or set the mailSender parameter.");
371         } else if (fromDeveloperId == null) {
372             final Developer dev = from.get(0);
373             return new MailSender(dev.getName(), dev.getEmail());
374         } else {
375             for (Developer developer : from) {
376                 if (fromDeveloperId.equals(developer.getId())) {
377                     return new MailSender(developer.getName(), developer.getEmail());
378                 }
379             }
380             throw new MojoExecutionException(
381                     "Missing developer with id '" + fromDeveloperId + "' in the <developers> section in your pom.");
382         }
383     }
